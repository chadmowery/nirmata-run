        </div>
      )}

    Add the `renderEditor` function inside the component:
    ```typescript
    function renderEditor(item: { section: Section; name: string }) {
      const isNew = item.name === '__new__';
      switch (item.section) {
        case 'entities':
          return <EntityTemplateEditor key={item.name} name={item.name} isNew={isNew} />;
        case 'mixins':
          return <MixinEditor key={item.name} name={item.name} isNew={isNew} />;
        case 'spawnTables':
          return <SpawnTableEditor key={item.name} name={item.name} isNew={isNew} />;
        case 'shells':
          return <ShellEditor key={item.name} name={item.name} isNew={isNew} />;
        case 'componentSchemas':
          return <ComponentSchemaViewer key={item.name} name={item.name} />;
        default:
          return <div className={styles.emptyState}><span>UNKNOWN_SECTION</span></div>;
      }
    }
    ```

    The `key={item.name}` prop is critical — it forces React to remount the editor when the user selects a different item, ensuring fresh state loads for each item.

    Also add the `Section` import from `./types` if not already present.

    Also load component schemas on mount into the store. Add this useEffect:
    ```typescript
    useEffect(() => {
      fetch('/api/dev/gamedata/components')
        .then(r => r.json())
        .then(data => {
          if (data.items) {
            setSectionItems('componentSchemas', data.items.map((s: { key: string }) => s.key));
          }
        })
        .catch(() => {});
    }, [setSectionItems]);
    ```

    This ensures the Component Schemas section in the sidebar is populated with the 38 component keys.

    NOTE: The store already has `cloneSource`, `setCloneSource`, and `clearCloneSource` (created in Plan 03). Do NOT modify store.ts — it is already complete.
