    ```typescript
    // Pre-load default loadout bundle (per D-09: every Shell comes pre-loaded)
    if (config.shellRecord) {
      const defaultBundle = getDefaultBundle(config.shellRecord.archetypeId);
      if (defaultBundle) {
        const firmwareEntityIds: number[] = [];
        const augmentEntityIds: number[] = [];

        // Create Firmware entities from bundle template IDs
        for (const firmwareTemplateId of defaultBundle.firmware) {
          try {
            const fwEntityId = entityFactory.create(world, firmwareTemplateId, componentRegistry);
            firmwareEntityIds.push(fwEntityId);
          } catch (e) {
            console.warn(`[ENGINE] Failed to create firmware entity from template '${firmwareTemplateId}':`, e);
          }
        }

        // Create Augment entities from bundle template IDs
        for (const augmentTemplateId of defaultBundle.augments) {
          try {
            const augEntityId = entityFactory.create(world, augmentTemplateId, componentRegistry);
            augmentEntityIds.push(augEntityId);
          } catch (e) {
            console.warn(`[ENGINE] Failed to create augment entity from template '${augmentTemplateId}':`, e);
          }
        }

        // Equip onto player's slots by updating FirmwareSlots and AugmentSlots components
        if (firmwareEntityIds.length > 0) {
          world.setComponent(placement.playerId, FirmwareSlots, { equipped: firmwareEntityIds });
        }
        if (augmentEntityIds.length > 0) {
          world.setComponent(placement.playerId, AugmentSlots, { equipped: augmentEntityIds });
        }
      }
    }
    ```
