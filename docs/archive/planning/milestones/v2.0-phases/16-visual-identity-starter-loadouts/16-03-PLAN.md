    **Step 1: Create `src/game/shells/loadout-bundles.ts`** (per D-08, D-09, D-10, D-11):

    ```typescript
    /**
     * Starter loadout bundle definitions.
     * Maps Shell archetypes to their default Firmware + Augment equipment.
     * Used for initial equipment and RESTORE DEFAULT functionality.
     */

    export interface LoadoutBundle {
      /** Shell archetype template ID (e.g., 'striker-v1') */
      shellId: string;
      /** Display name for the bundle (e.g., 'VANGUARD') */
      bundleName: string;
      /** Playstyle description tag */
      playstyleTag: string;
      /** Entity template IDs for default Firmware */
      firmware: string[];
      /** Entity template IDs for default Augments */
      augments: string[];
      /** Entity template IDs for default Software (empty for starters) */
      software: string[];
    }

    export const STARTER_BUNDLES: LoadoutBundle[] = [
      {
        shellId: 'striker-v1',
        bundleName: 'VANGUARD',
        playstyleTag: 'Aggressive Close-Quarters',
        firmware: ['phase-shift'],
        augments: ['displacement-venting'],
        software: [],
      },
      {
        shellId: 'bastion-v1',
        bundleName: 'OPERATOR',
        playstyleTag: 'Mid-Range Tactical Tank',
        firmware: ['neural-spike'],
        augments: ['static-siphon'],
        software: [],
      },
      {
        shellId: 'signal-v1',
        bundleName: 'GHOST',
        playstyleTag: 'Recon & Utility Scout',
        firmware: ['extended-sight'],
        augments: ['neural-feedback'],
        software: [],
      },
    ];

    /**
     * Get the default loadout bundle for a Shell archetype.
     */
    export function getDefaultBundle(shellId: string): LoadoutBundle | undefined {
      return STARTER_BUNDLES.find(b => b.shellId === shellId);
    }

    /**
     * Get the equipment template IDs to restore a Shell to its default loadout.
     * Returns null if no default bundle exists for the given Shell.
     */
    export function restoreDefaultLoadout(shellId: string): {
      firmware: string[];
      augments: string[];
      software: string[];
    } | null {
      const bundle = getDefaultBundle(shellId);
      if (!bundle) return null;
      return {
        firmware: [...bundle.firmware],
        augments: [...bundle.augments],
        software: [...bundle.software],
      };
    }
    ```

    ```typescript
    import { describe, it, expect } from 'vitest';
    import {
      STARTER_BUNDLES,
      getDefaultBundle,
      restoreDefaultLoadout,
    } from '../loadout-bundles';

    describe('LoadoutBundles', () => {
      describe('STARTER_BUNDLES', () => {
        it('has exactly 3 bundles', () => {
          expect(STARTER_BUNDLES).toHaveLength(3);
        });

        it('Vanguard: striker-v1 + phase-shift + displacement-venting', () => {
          const v = STARTER_BUNDLES.find(b => b.shellId === 'striker-v1');
          expect(v).toBeDefined();
          expect(v!.bundleName).toBe('VANGUARD');
          expect(v!.playstyleTag).toBe('Aggressive Close-Quarters');
          expect(v!.firmware).toContain('phase-shift');
          expect(v!.augments).toContain('displacement-venting');
        });

        it('Operator: bastion-v1 + neural-spike + static-siphon', () => {
          const o = STARTER_BUNDLES.find(b => b.shellId === 'bastion-v1');
          expect(o).toBeDefined();
          expect(o!.bundleName).toBe('OPERATOR');
          expect(o!.playstyleTag).toBe('Mid-Range Tactical Tank');
          expect(o!.firmware).toContain('neural-spike');
          expect(o!.augments).toContain('static-siphon');
        });

        it('Ghost: signal-v1 + extended-sight + neural-feedback', () => {
          const g = STARTER_BUNDLES.find(b => b.shellId === 'signal-v1');
          expect(g).toBeDefined();
          expect(g!.bundleName).toBe('GHOST');
          expect(g!.playstyleTag).toBe('Recon & Utility Scout');
          expect(g!.firmware).toContain('extended-sight');
          expect(g!.augments).toContain('neural-feedback');
        });

        it('all bundles have non-empty required fields', () => {
          for (const bundle of STARTER_BUNDLES) {
            expect(bundle.bundleName.length).toBeGreaterThan(0);
            expect(bundle.playstyleTag.length).toBeGreaterThan(0);
            expect(bundle.firmware.length).toBeGreaterThan(0);
            expect(bundle.augments.length).toBeGreaterThan(0);
          }
        });
      });

      describe('getDefaultBundle', () => {
        it('returns Vanguard for striker-v1', () => {
          const b = getDefaultBundle('striker-v1');
          expect(b?.bundleName).toBe('VANGUARD');
        });

        it('returns Operator for bastion-v1', () => {
          const b = getDefaultBundle('bastion-v1');
          expect(b?.bundleName).toBe('OPERATOR');
        });

        it('returns Ghost for signal-v1', () => {
          const b = getDefaultBundle('signal-v1');
          expect(b?.bundleName).toBe('GHOST');
        });

        it('returns undefined for nonexistent shell', () => {
          expect(getDefaultBundle('nonexistent')).toBeUndefined();
        });
      });

      describe('restoreDefaultLoadout', () => {
        it('returns equipment arrays for valid shell', () => {
          const result = restoreDefaultLoadout('striker-v1');
          expect(result).not.toBeNull();
          expect(result!.firmware).toEqual(['phase-shift']);
          expect(result!.augments).toEqual(['displacement-venting']);
          expect(result!.software).toEqual([]);
        });

        it('returns null for nonexistent shell', () => {
          expect(restoreDefaultLoadout('nonexistent')).toBeNull();
        });

        it('returns a copy (not reference) of arrays', () => {
          const r1 = restoreDefaultLoadout('striker-v1');
          const r2 = restoreDefaultLoadout('striker-v1');
          expect(r1!.firmware).not.toBe(r2!.firmware);
        });
      });
    });
    ```
