    import { Sprite, Container, Graphics } from 'pixi.js';
    import { AIBehaviorType } from '@shared/components/ai-state';

    interface Particle {
      graphics: Graphics;

    interface EmitterState {

      color: number;
      maxParticles: number;
      spawnTimer: number;
      spawnRate: number; // ms between spawns
    }

    Map AIBehaviorType to tier color:
    - `NULL_POINTER`, `BUFFER_OVERFLOW` -> 0x00F0FF (cyan)
    - `FRAGMENTER`, `LOGIC_LEAKER` -> 0xFF0055 (pink)

    - default -> 0x00F0FF (cyan)

    `attachNeonEmitter(sprite: Sprite, behaviorType: AIBehaviorType, effectsLayer: Container): void`
    - Creates a Container child on effectsLayer
    - Initializes EmitterState with maxParticles=15, spawnRate=100ms
    - Stores in `WeakMap<Sprite, EmitterState>`

    `updateAllEmitters(deltaMs: number): void`
    - For each emitter: update existing particles (move, fade, remove expired), spawn new if under max
    - Particles: 3x3 pixel rectangles, random velocity outward from sprite center, alpha fades from 0.8 to 0 over 600ms lifetime
    - Position particles relative to parent sprite position (read sprite.x, sprite.y)

    `cleanupEmitter(sprite: Sprite): void`
    - Destroy all particle graphics
    - Remove container from effectsLayer
    - Delete from WeakMap

    `disposeAllEmitters(): void`
    - Clean up all tracked emitters

    **Step 3: Create `src/components/ui/AugmentFlash.tsx`** (per VIS-05, D-05 discretion):

    New component that renders a geometric shape flash center-screen when an Augment triggers:

    ```tsx
    'use client';
    import React, { useEffect, useState } from 'react';
    import { useStore } from 'zustand';
    import { gameStore } from '@/game/ui/store';
    import styles from './AugmentFlash.module.css';

    export const AugmentFlash: React.FC = () => {
      const augmentFlashData = useStore(gameStore, (s) => s.augmentFlashData);
      const clearAugmentFlash = useStore(gameStore, (s) => s.clearAugmentFlash);
      const [visible, setVisible] = useState(false);

      useEffect(() => {
        if (augmentFlashData) {
          setVisible(true);
          const timer = setTimeout(() => {
            setVisible(false);
            clearAugmentFlash();
          }, 300);
          return () => clearTimeout(timer);
        }
      }, [augmentFlashData, clearAugmentFlash]);

      if (!visible || !augmentFlashData) return null;

      // Alternate between triangle (white) and square (orange) based on count
      const isTriangle = augmentFlashData.count % 2 === 1;

      return (
        <div className={styles.flashContainer}>
          <div className={isTriangle ? styles.triangle : styles.square} />
          <div className={styles.augmentName}>{augmentFlashData.augmentNames[0]}</div>
        </div>
      );
    };
    ```
